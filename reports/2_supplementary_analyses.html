run_segmented_analysis <- function(data, outcome_var, alternative = "two.sided", npsi = 1, seed = 100) {
  stopifnot(outcome_var %in% names(data))

  # Set seed for reproducibility
  set.seed(seed)

  # Formula
  formula <- as.formula(paste(outcome_var, "~ age"))

  # Fit linear and segmented models
  fit_lm <- lm(formula, data = data)
  fit_seg <- segmented::segmented(fit_lm, seg.Z = ~ age, npsi = npsi)

  # Extract breakpoint estimate and CI
  bp_est <- segmented::summary.segmented(fit_seg)$psi["psi1.age", "Est."]
  bp_ci <- confint.segmented(fit_seg)["psi1.age", c("CI(95%).low", "CI(95%).up")]

  # Slope estimates before and after breakpoint
  slopes <- segmented::slope(fit_seg)$age

  # Davies test for slope difference
  pscore <- segmented::pscore.test(fit_seg, seg.Z = ~ age, k = 10, alternative = alternative)$p.value

  # Tidy output
  tibble::tibble(
    outcome = outcome_var,
    breakpoint = round(bp_est, 2),
    ci_lower = round(bp_ci[1], 2),
    ci_upper = round(bp_ci[2], 2),
    slope_before = round(slopes[1], 4),
    slope_after = round(slopes[2], 4),
    slope_diff = round(slopes[2] - slopes[1], 4),
    pscore_pval = signif(pscore, 3),
    model = list(fit_seg)
  )
}

Define a function for reproducing breakpoint plots

make_breakpoint_plot <- function(data,
                                 age_var,
                                 y_var,
                                 fit_object,
                                 breakpoint_coords = NULL,
                                 secondary_breaks = NULL,
                                 plot_tag = NULL,
                                 title = NULL,
                                 subtitle = NULL,
                                 y_label = NULL,
                                 ribbon_color = "black",
                                 line_color = "blue") {
  # Extract fitted values and confidence intervals from segmented model
  fit_vals <- broken.line(fit_object)$fit
  se_vals  <- broken.line(fit_object, se.fit = TRUE)$se.fit
  ribbon_low  <- fit_vals - 1.96 * se_vals
  ribbon_high <- fit_vals + 1.96 * se_vals

  # Auto-calculate breakpoint_coords if not provided
  if (is.null(breakpoint_coords)) {
    bp_x <- tryCatch(segmented::summary.segmented(fit_object)$psi["psi1.age", "Est."], error = function(e) NA)
    bp_y <- tryCatch(predict(fit_object, newdata = data.frame(age = bp_x)), error = function(e) NA)
    if (!is.na(bp_x) && !is.na(bp_y)) {
      breakpoint_coords <- c(bp_x, bp_y)
    }
  }

  # Try to auto-extract CI bounds if not supplied
  if (is.null(secondary_breaks)) {
    secondary_breaks <- tryCatch(confint.segmented(fit_object)["psi1.age", c("CI(95%).low", "CI(95%).up")], error = function(e) NULL)
  }

  # Begin ggplot
  p <- ggplot(data, aes_string(x = age_var, y = y_var)) +

    # Add dot at estimated breakpoint
    {
      if (!is.null(breakpoint_coords)) {
        geom_point(data = NULL,
                   aes(x = breakpoint_coords[1], y = breakpoint_coords[2]),
                   shape = 19, colour = "black", size = 4)
      }
    } +

    # Add fitted line
    geom_line(aes_string(x = age_var, y = "fit_vals"),
              linewidth = 2, colour = line_color, alpha = 0.65) +

    # Add confidence ribbon
    geom_ribbon(aes_string(
      x = age_var,
      y = "fit_vals",
      ymin = "ribbon_low",
      ymax = "ribbon_high"
    ),
    fill = ribbon_color, alpha = 0.15) +

    # Raw data points colored by sex
    geom_point(aes(color = Sex), size = 2, alpha = 0.65) +

    # Add vertical lines at CI bounds
    {
      if (!is.null(secondary_breaks)) {
        geom_vline(data = data.frame(x = secondary_breaks),
                   aes(xintercept = x),
                   linetype = "dotted", colour = "black", size = 1, alpha = 0.5)
      }
    } +

    # Labels
    labs(
      tag = plot_tag,
      title = title,
      subtitle = subtitle,
      x = "Age (years)",
      y = y_label
    ) +

    # Theme
    theme(
      axis.title     = element_text(size = 14),
      plot.title     = element_text(face = "bold", size = 16),
      plot.subtitle  = element_text(size = 14)
    )

  return(p)
}

Create wrapper function to reproduce piecewise sex plots

make_dual_breakpoint_plot <- function(data_m,
                                      data_f,
                                      age_var,
                                      y_var,
                                      fit_m,
                                      fit_f,
                                      break_m = NULL,        # coordinate: c(x, y)
                                      break_f = NULL,        # coordinate: c(x, y)
                                      break_ci_m = NULL,
                                      break_ci_f = NULL,
                                      plot_tag = NULL,
                                      title = NULL,
                                      subtitle = NULL,
                                      y_label = NULL,
                                      point_size = 2,
                                      ribbon_alpha = 0.15,
                                      line_alpha = 0.65,
                                      colors = c(male = "blue", female = "red")) {

  # Fit predictions
  fit_vals_m     <- broken.line(fit_m)$fit
  se_vals_m      <- broken.line(fit_m, se.fit = TRUE)$se.fit
  ribbon_low_m   <- fit_vals_m - 1.96 * se_vals_m
  ribbon_high_m  <- fit_vals_m + 1.96 * se_vals_m

  fit_vals_f     <- broken.line(fit_f)$fit
  se_vals_f      <- broken.line(fit_f, se.fit = TRUE)$se.fit
  ribbon_low_f   <- fit_vals_f - 1.96 * se_vals_f
  ribbon_high_f  <- fit_vals_f + 1.96 * se_vals_f

  # Get p-values for slope difference
  pval_m <- tryCatch(pscore.test(fit_m, seg.Z = ~age, k = 10)$p.value, error = function(e) NA)
  pval_f <- tryCatch(pscore.test(fit_f, seg.Z = ~age, k = 10)$p.value, error = function(e) NA)

  sig_m <- !is.na(pval_m) && pval_m < 0.05
  sig_f <- !is.na(pval_f) && pval_f < 0.05

  # Auto breakpoints
  if (is.null(break_m) && sig_m) {
    bp_x_m <- tryCatch(summary(fit_m)$psi["psi1.age", "Est."], error = function(e) NA)
    bp_y_m <- tryCatch(predict(fit_m, newdata = data.frame(age = bp_x_m)), error = function(e) NA)
    if (!is.na(bp_x_m) && !is.na(bp_y_m)) break_m <- c(bp_x_m, bp_y_m)
  }

  if (is.null(break_f) && sig_f) {
    bp_x_f <- tryCatch(summary(fit_f)$psi["psi1.age", "Est."], error = function(e) NA)
    bp_y_f <- tryCatch(predict(fit_f, newdata = data.frame(age = bp_x_f)), error = function(e) NA)
    if (!is.na(bp_x_f) && !is.na(bp_y_f)) break_f <- c(bp_x_f, bp_y_f)
  }

  # Auto CI bounds
  if (is.null(break_ci_m) && sig_m) {
    break_ci_m <- tryCatch(confint.segmented(fit_m)["psi1.age", c("CI(95%).low", "CI(95%).up")], error = function(e) NULL)
  }

  if (is.null(break_ci_f) && sig_f) {
    break_ci_f <- tryCatch(confint.segmented(fit_f)["psi1.age", c("CI(95%).low", "CI(95%).up")], error = function(e) NULL)
  }

  # Begin plot
  p <- ggplot() +

    # Male breakpoint dot
    {
      if (sig_m && !is.null(break_m)) {
        geom_point(data = NULL,
                   aes(x = break_m[1], y = break_m[2]),
                   shape = 21, size = 5, fill = "black", color = "black")
      }
    } +

    # Female breakpoint dot
    {
      if (sig_f && !is.null(break_f)) {
        geom_point(data = NULL,
                   aes(x = break_f[1], y = break_f[2]),
                   shape = 21, size = 5, fill = "black", color = "black")
      }
    } +

    # Raw points
    geom_point(data = data_m, aes_string(x = age_var, y = y_var),
               colour = colors["male"], size = point_size, alpha = 0.5) +
    geom_point(data = data_f, aes_string(x = age_var, y = y_var),
               colour = colors["female"], size = point_size, alpha = 0.5) +

    # Fitted lines
    geom_line(data = data_m, aes_string(x = age_var, y = "fit_vals_m"),
              linewidth = 2, colour = colors["male"], alpha = line_alpha) +
    geom_line(data = data_f, aes_string(x = age_var, y = "fit_vals_f"),
              linewidth = 2, colour = colors["female"], alpha = line_alpha) +

    # Confidence ribbons
    geom_ribbon(data = data_m,
                aes_string(x = age_var, y = "fit_vals_m",
                           ymin = "ribbon_low_m", ymax = "ribbon_high_m"),
                fill = colors["male"], alpha = ribbon_alpha) +
    geom_ribbon(data = data_f,
                aes_string(x = age_var, y = "fit_vals_f",
                           ymin = "ribbon_low_f", ymax = "ribbon_high_f"),
                fill = colors["female"], alpha = ribbon_alpha) +

    # CI lines for males
    {
      if (sig_m && !is.null(break_ci_m)) {
        geom_vline(data = data.frame(x = break_ci_m),
                   aes(xintercept = x),
                   linetype = "dotted", color = colors["male"],
                   size = 1, alpha = 0.5)
      }
    } +

    # CI lines for females
    {
      if (sig_f && !is.null(break_ci_f)) {
        geom_vline(data = data.frame(x = break_ci_f),
                   aes(xintercept = x),
                   linetype = "dotted", color = colors["female"],
                   size = 1, alpha = 0.5)
      }
    } +

    labs(
      tag = plot_tag,
      title = title,
      subtitle = subtitle,
      x = "Age (years)",
      y = y_label
    ) +

    theme(
      plot.title    = element_markdown(lineheight = 1.1, size = 16),
      legend.text   = element_markdown(size = 11),
      plot.caption  = element_markdown(size = 9, hjust = 0.5),
      axis.title    = element_text(size = 12)
    )

  return(p)
}

Define a reusable function to specify SEM models and fit them using lavaan::sem()

serial_med <- function(x, m1, m2, y, cov, dat, boot = 5000, seeder = 100) {
  
  set.seed(seeder)
  
  # Ensure the model string is properly formatted using str_glue
  med.model <- str_glue("
    # Regressions
    {m1} ~ a1*{x} + cov1*{cov}  # Label the a coefficient in the M1 regression.
    {m2} ~ a2*{x} + d21*{m1} + cov2*{cov}  # Label the a coefficientS in the M2 regression.
    {y} ~ cPrime*{x} + b1*{m1} + b2*{m2} + cov3*{cov}

    # Define New Parameters (products of coefficients)
    ab1 := a1 * b1
    ab2 := a2 * b2
    ab3 := a1 * d21 * b2
    c := cPrime + ab1 + ab2 + ab3
  ", .sep = "")

 # Fit the model using lavaan's sem function
  fit <- lavaan::sem(model = med.model, data = dat, test = "bootstrap", bootstrap = boot)

  # Extract parameter estimates, including bootstrapping if needed
  param_est <- parameterEstimates(fit, standardized = TRUE, level = 0.95, boot.ci.type = "bca.simple")
  
  #  Tidy output
  tibble::tibble(
    parameter = c("a1*b1", "a2*b2", "a1*d21*b2"),
    coef_std = c(param_est[param_est$label == "ab1", "std.all"],
                 param_est[param_est$label == "ab2", "std.all"],
                 param_est[param_est$label == "ab3", "std.all"]),
    ci_lower = c(param_est[param_est$label == "ab1", "ci.lower"],
                 param_est[param_est$label == "ab2", "ci.lower"],
                 param_est[param_est$label == "ab3", "ci.lower"]),
    ci_upper = c(param_est[param_est$label == "ab1", "ci.upper"],
                 param_est[param_est$label == "ab2", "ci.upper"],
                 param_est[param_est$label == "ab3", "ci.upper"]),
    p_value_uncorrected = c(param_est[param_est$label == "ab1", "pvalue"],
                 param_est[param_est$label == "ab2", "pvalue"],
                 param_est[param_est$label == "ab3", "pvalue"])
  )
}
